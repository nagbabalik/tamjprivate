$uid = 10684

while ($true) {
    Clear-Host
    Write-Host "Watching MobiVPN connections for UID $uid"
    Write-Host "Connect MobiVPN now. Press Ctrl+C to stop."
    Write-Host ""

    $results = @()
    $lines = adb shell "cat /proc/net/tcp 2>/dev/null"

    foreach ($line in ($lines | Select-Object -Skip 1)) {
        $fields = $line.Trim() -split '\s+'

        if ($fields.Count -le 7) {
            continue
        }

        if ([int]$fields[7] -ne $uid) {
            continue
        }

        $remote = $fields[2] -split ":"
        $hexIp = $remote[0]
        $hexPort = $remote[1]

        if ($hexIp.Length -ne 8) {
            continue
        }

        if ($hexIp -eq "00000000") {
            continue
        }

        $octets = @()

        for ($i = 0; $i -lt 4; $i++) {
            $octets += [Convert]::ToInt32(
                $hexIp.Substring($i * 2, 2),
                16
            )
        }

        [array]::Reverse($octets)

        $ip = $octets -join "."
        $port = [Convert]::ToInt32($hexPort, 16)
        $state = $fields[3]

        $results += "$ip`:$port state=$state"
    }

    $results | Sort-Object -Unique
    Start-Sleep -Milliseconds 500
}
